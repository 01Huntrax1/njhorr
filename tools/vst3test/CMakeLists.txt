cmake_minimum_required(VERSION 3.16)
project(VstTest VERSION 1.0 LANGUAGES CXX C)
if (NOT JUCE_PATH)
    message (FATAL_ERROR "set JUCE_PATH")
endif()
add_subdirectory(${JUCE_PATH} juce)
set(CMAKE_CXX_STANDARD 17)
juce_add_plugin(TestGain FORMATS VST3 PRODUCT_NAME "TestGain" PLUGIN_CODE TgGn PLUGIN_MANUFACTURER_CODE Njts IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE)
target_sources(TestGain PRIVATE gain.cpp)
target_compile_definitions(TestGain PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(TestGain PRIVATE juce::juce_audio_utils)
juce_add_plugin(TestSine FORMATS VST3 PRODUCT_NAME "TestSine" PLUGIN_CODE TsSn PLUGIN_MANUFACTURER_CODE Njts IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE)
target_sources(TestSine PRIVATE sine.cpp)
target_compile_definitions(TestSine PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(TestSine PRIVATE juce::juce_audio_utils)
juce_add_console_app(HostTest PRODUCT_NAME "HostTest")
target_sources(HostTest PRIVATE hosttest.cpp)
target_compile_definitions(HostTest PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_USE_MP3AUDIOFORMAT=0)
target_include_directories(HostTest PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../include/cmajor/helpers ${CMAKE_CURRENT_SOURCE_DIR}/../../native_ui)
target_link_libraries(HostTest PRIVATE juce::juce_audio_utils juce::juce_gui_basics)
