cmake_minimum_required(VERSION 3.16)
project(NjoUiTest VERSION 1.0 LANGUAGES CXX C)
if (NOT JUCE_PATH)
    message(FATAL_ERROR "set -DJUCE_PATH=<JUCE folder>")
endif()
add_subdirectory(${JUCE_PATH} juce)
file(GLOB UI_SOURCES ${CMAKE_CURRENT_SOURCE_DIR}/../*.cpp ${CMAKE_CURRENT_SOURCE_DIR}/../pages/*.cpp)
juce_add_console_app(njo_shot PRODUCT_NAME "njo_shot")
target_sources(njo_shot PRIVATE ${UI_SOURCES} shot.cpp)
target_compile_features(njo_shot PRIVATE cxx_std_17)
target_compile_definitions(njo_shot PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_USE_FLAC=1 JUCE_USE_OGGVORBIS=1)
target_link_libraries(njo_shot PRIVATE juce::juce_gui_basics juce::juce_audio_formats juce::juce_audio_basics juce::juce_events juce::juce_graphics juce::juce_data_structures juce::juce_recommended_config_flags)
